Security note: This HTML runs in the visitor’s browser, so its source can be viewed or copied. The main cloud record is configured without a delete permission for normal signed-in website sessions.

One-time cloud database setup (included in this single HTML file)

Create a Supabase project, open its SQL Editor, and run the SQL below once. Then replace the two Supabase placeholders near the beginning of the main JavaScript section in this HTML file with your Project URL and public publishable/anon key. Do not use a service-role or secret key.

In Supabase Authentication, enable Email OTP and use {{ .Token }} in the email template so a six-digit code is sent to the authorized mailbox.

-- Hemanth Study Store: private cloud storage setup
-- Run this entire script in Supabase SQL Editor for your project.

create table if not exists public.hemanth_study_store (
  user_id uuid primary key references auth.users(id) on delete cascade,
  owner_email text not null default 'kvssmh464@gmail.com',
  payload jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now(),
  constraint hemanth_study_store_owner_email_check
    check (lower(owner_email) = 'kvssmh464@gmail.com')
);

alter table public.hemanth_study_store enable row level security;

revoke all on table public.hemanth_study_store from anon;
revoke delete, truncate, references, trigger on table public.hemanth_study_store from authenticated;
grant select, insert, update on table public.hemanth_study_store to authenticated;

-- Remove any older delete policy: normal website sessions cannot delete the main cloud record.
drop policy if exists "Hemanth can delete own Study Store" on public.hemanth_study_store;
drop policy if exists "Hemanth can read own Study Store" on public.hemanth_study_store;
drop policy if exists "Hemanth can create own Study Store" on public.hemanth_study_store;
drop policy if exists "Hemanth can update own Study Store" on public.hemanth_study_store;

create policy "Hemanth can read own Study Store"
on public.hemanth_study_store for select to authenticated
using (
  auth.uid() = user_id
  and lower(coalesce(auth.jwt() ->> 'email', '')) = 'kvssmh464@gmail.com'
);

create policy "Hemanth can create own Study Store"
on public.hemanth_study_store for insert to authenticated
with check (
  auth.uid() = user_id
  and lower(coalesce(auth.jwt() ->> 'email', '')) = 'kvssmh464@gmail.com'
  and lower(owner_email) = 'kvssmh464@gmail.com'
);

create policy "Hemanth can update own Study Store"
on public.hemanth_study_store for update to authenticated
using (
  auth.uid() = user_id
  and lower(coalesce(auth.jwt() ->> 'email', '')) = 'kvssmh464@gmail.com'
)
with check (
  auth.uid() = user_id
  and lower(coalesce(auth.jwt() ->> 'email', '')) = 'kvssmh464@gmail.com'
  and lower(owner_email) = 'kvssmh464@gmail.com'
);

-- No DELETE policy is created for the main Study Store record.

-- Daily recovery snapshots: one backup per day, per authenticated owner.
create table if not exists public.hemanth_study_store_backups (
  user_id uuid not null references auth.users(id) on delete cascade,
  backup_date date not null default (now() at time zone 'utc')::date,
  payload jsonb not null,
  created_at timestamptz not null default now(),
  primary key (user_id, backup_date),
  constraint hemanth_study_store_backups_owner_check
    check (user_id is not null)
);

alter table public.hemanth_study_store_backups enable row level security;
revoke all on table public.hemanth_study_store_backups from anon;
grant select, insert, delete on table public.hemanth_study_store_backups to authenticated;

drop policy if exists "Hemanth can read own Study Store backups" on public.hemanth_study_store_backups;
drop policy if exists "Hemanth can create own Study Store backups" on public.hemanth_study_store_backups;
drop policy if exists "Hemanth can delete own Study Store backups" on public.hemanth_study_store_backups;

create policy "Hemanth can read own Study Store backups"
on public.hemanth_study_store_backups for select to authenticated
using (
